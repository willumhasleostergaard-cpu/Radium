<script>

/* ELEMENTS */

const messages = document.getElementById("messages");

const input =
    document.getElementById("messageInput");

const sendButton =
    document.getElementById("sendButton");

const imageInput =
    document.getElementById("imageInput");

const voiceButton =
    document.getElementById("voiceButton");


/* REMOVE WELCOME */

function removeWelcome() {

    const welcome =
        document.getElementById("welcome");

    if (welcome) {
        welcome.remove();
    }

}


/* SEND MESSAGE */

function sendMessage() {

    const text =
        input.value.trim();

    if (!text) return;

    removeWelcome();


    /* USER */

    const userMessage =
        document.createElement("div");

    userMessage.className =
        "message user";

    userMessage.textContent =
        text;

    messages.appendChild(userMessage);


    input.value = "";


    /* AI */

    const aiMessage =
        document.createElement("div");

    aiMessage.className =
        "message assistant";

    messages.appendChild(aiMessage);


    /* TEST RESPONSE */

    const response =
        "Hej! Jeg er Radium 🤖 " +
        "Dette svar bliver streamet " +
        "bogstav for bogstav.";

    let index = 0;


    const timer =
        setInterval(() => {

            aiMessage.textContent +=
                response[index];

            index++;

            messages.scrollTop =
                messages.scrollHeight;


            if (index >= response.length) {

                clearInterval(timer);

            }

        }, 35);

}


/* SEND BUTTON */

sendButton.addEventListener(
    "click",
    sendMessage
);


/* ENTER */

input.addEventListener(
    "keydown",
    function(event) {

        if (
            event.key === "Enter" &&
            !event.shiftKey
        ) {

            event.preventDefault();

            sendMessage();

        }

    }
);


/* IMAGE */

imageInput.addEventListener(
    "change",
    function() {

        const file =
            imageInput.files[0];

        if (!file) return;

        removeWelcome();


        const message =
            document.createElement("div");

        message.className =
            "message user";


        const image =
            document.createElement("img");

        image.className =
            "message-image";

        image.src =
            URL.createObjectURL(file);


        message.appendChild(image);

        messages.appendChild(message);


        messages.scrollTop =
            messages.scrollHeight;


        imageInput.value = "";

    }
);


/* NEW CHAT */

function newChat() {

    messages.innerHTML = `

        <div
            class="welcome"
            id="welcome">

            <h1>
                Hvad kan jeg hjælpe med?
            </h1>

            <p>
                Velkommen til Radium AI.
            </p>

        </div>

    `;

}


/* VOICE */

voiceButton.addEventListener(
    "click",
    function() {

        if (
            !(
                "webkitSpeechRecognition"
                in window
            )
        ) {

            alert(
                "Voice understøttes ikke " +
                "af denne browser."
            );

            return;

        }


        const recognition =
            new webkitSpeechRecognition();


        recognition.lang =
            "da-DK";


        recognition.start();


        recognition.onresult =
            function(event) {

                input.value =
                    event.results[0][0]
                    .transcript;

            };

    }
);
